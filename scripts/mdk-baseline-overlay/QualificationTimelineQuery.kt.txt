package dev.ipf.whitenoise.android.qualification

import dev.ipf.marmotkit.TimelineMessageQueryFfi

/** The newest [limit] messages of [group] on the baseline runtime, whose query predates the filter fields. */
internal fun qualificationTimelineQuery(
    group: String,
    limit: UInt,
) = TimelineMessageQueryFfi(
    groupIdHex = group,
    search = null,
    before = null,
    beforeMessageId = null,
    after = null,
    afterMessageId = null,
    limit = limit,
)
